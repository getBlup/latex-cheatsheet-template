% Page number formatting
\AddToHook{shipout/foreground}{%
  \begin{tikzpicture}[remember picture,overlay]
    \node[
      anchor=north east,
      xshift=-0.2cm,
      yshift=-0.2cm,
      draw,
      circle,
      fill=yellow,
      inner sep=2pt
    ] at (current page.north east) {\thepage};
  \end{tikzpicture}%
}

% Formatting for tableofcontents
\setlength{\cftbeforesecskip}{2pt}
\setlength{\cftbeforesubsecskip}{0pt}

% Formatting for columns
\setlength{\columnseprule}{0.4pt}
\setlength{\columnsep}{2mm}
\setlength{\parindent}{0pt}

% Section formatting
\titleformat{\section} % which level
  {\sffamily\small\bfseries} % font, color, size, weight
  {\thesection} % section number
  {1em} % space between number and title
  {} % before the title text itself

% Spacing before/after sections
\titlespacing{\section}{0pt}{2ex}{1ex} % {left}{before}{after}

%Subsection formatting
\titleformat{\subsection}
  {\sffamily\bfseries} % no color, bold, sans-serif
  {\thesubsection} % number
  {1em} % number-title separation
  {} % before title text

% Spacing before/after subsections
\titlespacing{\subsection}{0pt}{1.5ex}{0.5ex} % left, before, after
